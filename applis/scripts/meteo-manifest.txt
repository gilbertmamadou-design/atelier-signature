# ---------- 1. Dans <application> … </application> ----------
<service android:name=".MeteoPersoJob" android:permission="android.permission.BIND_JOB_SERVICE" android:exported="false" />

# ---------- 2. Avant </manifest> : autorisations ----------
# Vérification périodique avec contrainte réseau (JobScheduler l'exige pour setRequiredNetworkType)
<uses-permission android:name="android.permission.ACCESS_NETWORK_STATE" />
# Tâche conservée après un redémarrage du téléphone (setPersisted)
<uses-permission android:name="android.permission.RECEIVE_BOOT_COMPLETED" />
# Internet : déjà présent dans le manifeste Capacitor
<uses-permission android:name="android.permission.INTERNET" />
# Déjà ajoutées pour toutes les applis MAG par print-patch.sh (à garder) :
#   ACCESS_COARSE_LOCATION (position approximative : suffit pour la météo), ACCESS_FINE_LOCATION, POST_NOTIFICATIONS,
#   fr.magformation13.permission.WIDGET (tuile Socle et canal de sauvegarde)
# Pas besoin d'ACCESS_BACKGROUND_LOCATION : les alertes surveillent un lieu fixe (ville choisie,
# ou dernière position connue enregistrée quand l'appli était ouverte).

# ---------- 3. Notes ----------
