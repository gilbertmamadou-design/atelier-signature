# Écran : lignes à ajouter à android/app/src/main/AndroidManifest.xml (uniquement pour l'appli Écran)
# Copier aussi scripts/EcranPlayer.java à côté de MainActivity.java (avec __PKG__ remplacé par le paquet), comme les *Bridge.java.

# --- 1) Avant </manifest> ---
    <uses-permission android:name="android.permission.INTERNET" />
    <uses-permission android:name="android.permission.READ_MEDIA_VIDEO" />
    <uses-permission android:name="android.permission.READ_EXTERNAL_STORAGE" android:maxSdkVersion="32" />
    <queries>
        <package android:name="com.google.android.youtube" />
        <package android:name="fr.francetv.pluzz" />
        <package android:name="tv.arte.plus7" />
        <package android:name="fr.tf1.mytf1" />
        <package android:name="fr.m6.m6replay" />
        <package android:name="com.dotscreen.tv5mondeplus.mobile" />
        <package android:name="fr.artestudio.arteradio.mobile" />
        <package android:name="fr.magformation13.ondes" />
        <package android:name="fr.magformation13.galerie" />
        <intent><action android:name="android.intent.action.VIEW" /><data android:scheme="market" /></intent>
        <intent><action android:name="android.intent.action.VIEW" /><data android:scheme="https" /></intent>
        <intent><action android:name="android.intent.action.VIEW" /><data android:mimeType="video/*" /></intent>
    </queries>

# --- 2) Avant </application> ---
        <activity android:name=".EcranPlayer"
            android:label="Écran"
            android:exported="true"
            android:theme="@android:style/Theme.Black.NoTitleBar.Fullscreen"
            android:screenOrientation="sensorLandscape"
            android:configChanges="orientation|screenSize|smallestScreenSize|screenLayout|keyboard|keyboardHidden|navigation|uiMode|density"
            android:supportsPictureInPicture="true"
            android:resizeableActivity="true"
            android:launchMode="singleTask"
            android:taskAffinity="ecran.lecteur"
            android:excludeFromRecents="false"
            android:autoRemoveFromRecents="true">
            <intent-filter android:label="Ouvrir avec Écran">
                <action android:name="android.intent.action.VIEW" />
                <category android:name="android.intent.category.DEFAULT" />
                <data android:scheme="content" android:mimeType="video/*" />
                <data android:scheme="file" android:mimeType="video/*" />
            </intent-filter>
            <intent-filter android:label="Ouvrir avec Écran">
                <action android:name="android.intent.action.VIEW" />
                <category android:name="android.intent.category.DEFAULT" />
                <category android:name="android.intent.category.BROWSABLE" />
                <data android:scheme="http" android:mimeType="video/*" />
                <data android:scheme="https" android:mimeType="video/*" />
                <data android:scheme="https" android:mimeType="application/x-mpegURL" />
                <data android:scheme="https" android:mimeType="application/vnd.apple.mpegurl" />
            </intent-filter>
        </activity>

# Remarques
# - android:screenOrientation n'est qu'une valeur de départ : EcranPlayer la change selon le format de la vidéo
#   (portrait ou paysage) ou selon le réglage de rotation choisi dans le lecteur.
# - taskAffinity + singleTask : le lecteur a sa propre tâche, nécessaire au mode image dans l'image
#   (le retour depuis la fenêtre flottante ne ramène pas la page Écran au premier plan par erreur).
# - Les noms de paquets des applis TV ont été vérifiés sur le Play Store (fiches publiques) ;
#   si l'un change, la page bascule sur une recherche Play Store par nom.
