# Coffre (gestionnaire de mots de passe) : lignes à ajouter à android/app/src/main/AndroidManifest.xml (uniquement pour l'appli Coffre)
# CoffreBridge.java est déjà copié automatiquement (boucle scripts/*Bridge.java) et chargé sous window.MAGCoffre.
# Aucun autre fichier natif à copier.

# ---------- 1. Avant </manifest> : empreinte ----------
<uses-permission android:name="android.permission.USE_BIOMETRIC" />
<uses-permission android:name="android.permission.USE_FINGERPRINT" android:maxSdkVersion="28" />
# (USE_FINGERPRINT ne sert qu'à Android 9, pour savoir si une empreinte est enregistrée.)

# ---------- 2. Dans <application ...> : pas de copie des données par Google ----------
# Le coffre est chiffré, mais il n'a rien à faire dans une sauvegarde Google Drive ni dans un transfert d'appareil
# (la clé d'empreinte du Keystore ne suit pas de toute façon). Ajouter ces attributs à la balise <application> :
#     android:allowBackup="false"
#     android:fullBackupContent="false"
# Si le manifeste porte déjà android:allowBackup="true", le remplacer par "false".

# ---------- 3. Facultatif : supprimer l'accès internet ----------
# La page est déjà bloquée par sa politique de sécurité (Content-Security-Policy : connect-src 'none').
# Pour aller plus loin, retirer la permission INTERNET ajoutée par le modèle Capacitor
# (les pages sont servies en local par Capacitor, sans réseau). Nécessite xmlns:tools="http://schemas.android.com/tools"
# sur la balise <manifest> :
#     <uses-permission android:name="android.permission.INTERNET" tools:node="remove" />
# À tester sur le téléphone avant de l'adopter : si la page reste blanche, enlever cette ligne.
